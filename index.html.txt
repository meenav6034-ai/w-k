  function repImg(){
    var R=_rpd;if(!R)return;
    var W=760,P=24,RH=32,sc=2;
    var wt=R.cols.map(function(c,i){return i==0?0.5:/Vehicle/.test(c)?1.6:/Financer/.test(c)?1.8:/Bank|Debited/i.test(c)?2.2:1.4;});
    var sm=wt.reduce(function(a,b){return a+b;},0),cw=wt.map(function(w){return (W-2*P)*w/sm;});
    var rows=0;R.groups.forEach(function(g){if(g.h)rows++;rows+=g.rows.length;});
    var H=130+(rows+1)*RH+70;

    var canvas=document.createElement("canvas");canvas.width=W*sc;canvas.height=H*sc;
    var x=canvas.getContext("2d");x.scale(sc,sc);
    x.fillStyle="#ffffff";x.fillRect(0,0,W,H);
    x.fillStyle="#0b6e4f";x.fillRect(0,0,W,80);
    x.fillStyle="#ffffff";x.font="bold 24px sans-serif";x.fillText(R.title,P,36);
    x.font="13px sans-serif";x.fillText(R.fl+" · As on "+R.t0,P,60);

    function fit(t,m){
      t=String(t);if(x.measureText(t).width<=m)return t;
      while(t.length>1&&x.measureText(t+"…").width>m)t=t.slice(0,-1);
      return t+"…";
    }

    var y=95;
    function rowAt(ry,cells,o){
      if(o.bg){x.fillStyle=o.bg;x.fillRect(P,ry,W-2*P,RH);}
      x.strokeStyle="#e2e8f0";x.strokeRect(P,ry,W-2*P,RH);
      x.fillStyle=o.col||"#111";x.font=(o.b?"bold ":"")+"13px sans-serif";
      var cx=P;
      cells.forEach(function(t,i){
        var m=cw[i]-12;
        if(R.al[i]=="r"){x.textAlign="right";x.fillText(fit(t,m),cx+cw[i]-6,ry+21);}
        else if(R.al[i]=="c"){x.textAlign="center";x.fillText(fit(t,m),cx+cw[i]/2,ry+21);}
        else{x.textAlign="left";x.fillText(fit(t,m),cx+6,ry+21);}
        cx+=cw[i];
      });
      x.textAlign="left";
    }

    rowAt(y,R.cols,{bg:"#24303e",col:"#ffffff",b:1});y+=RH;
    R.groups.forEach(function(g){
      if(g.h){
        x.fillStyle="#e6f4ea";x.fillRect(P,y,W-2*P,RH);
        x.strokeStyle="#e2e8f0";x.strokeRect(P,y,W-2*P,RH);
        x.fillStyle="#0b6e4f";x.font="bold 13px sans-serif";
        x.fillText(g.h,P+8,y+21);
        y+=RH;
      }
      g.rows.forEach(function(r,idx){
        rowAt(y,r.c,{bg:idx%2?"#f8fafc":"#ffffff",col:r.red?"#dc2626":"#111111"});
        y+=RH;
      });
    });

    x.fillStyle="#111111";x.font="bold 13px sans-serif";x.fillText(R.foot,P,y+32);

    var imgUrl=canvas.toDataURL("image/png");
