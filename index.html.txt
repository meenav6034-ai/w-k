function initStorage() {

}

function uid() {
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 5);
}

function fmtDate(d) {
  var m = String(d.getMonth() + 1).padStart(2, "0");
  var dt = String(d.getDate()).padStart(2, "0");
  return d.getFullYear() + "-" + m + "-" + dt;
}

function offsetDate(n) {
  var d = new Date();
  d.setDate(d.getDate() + n);
  return fmtDate(d);
}

function diffDays(dStr) {
  var t = new Date(dStr + "T00:00:00").getTime();
  var now = new Date().setHours(0,0,0,0);
  return Math.round((t - now) / 864e5);
}

function addMonth(dStr) {
  var d = new Date(dStr + "T00:00:00");
  d.setMonth(d.getMonth() + 1);
  return fmtDate(d);
}

function inr(n) {
  return "₹" + Number(n || 0).toLocaleString("en-IN");
}

function getUnit(c) {
  for (var i = 0; i < CATS.length; i++) {
    if (CATS[i][0] == c) return CATS[i][2];
  }
  return "km";
}

function getIcon(c) {
  for (var i = 0; i < CATS.length; i++) {
    if (CATS[i][0] == c) return CATS[i][1];
  }
  return "🚘";
}

function getVeh(id) {
  for (var i = 0; i < DB.V.length; i++) {
    if (DB.V[i].id == id) return DB.V[i];
  }
  return { id: id, no: "-", nm: "-", c: "Truck", rd: 0, f: "" };
}

function getAlerts() {
  var list = [];
  DB.D.forEach(function(d) {
    if (!d.ex) return;
    var k = diffDays(d.ex);
    if (k <= 30) {
      list.push({
        type: "d", id: d.id, vid: d.vid, k: d.k, ex: d.ex,
        lv: k < 0 ? 0 : (k <= 7 ? 1 : 2),
        msg: d.k + (k < 0 ? " expired " + (-k) + " days ago" : (k == 0 ? " expires today" : " expires in " + k + " days"))
      });
    }
  });

  DB.E.forEach(function(e) {
    if (!e.due) return;
    var k = diffDays(e.due);
    if (k <= 5) {
      list.push({
        type: "e", id: e.id, vid: e.vid, ln: e.ln, amt: e.amt, due: e.due, bk: e.bk,
        lv: k < 0 ? 0 : 1,
        msg: "EMI Rs " + e.amt + " (" + e.ln + ") " + (k < 0 ? "overdue by " + (-k) + " days" : (k == 0 ? "due today" : "due in " + k + " days"))
      });
    }
  });

  DB.Sv.forEach(function(s) {
    var v = getVeh(s.vid);
    var u = getUnit(v.c);
    var nextAt = (s.last || 0) + (s.int || 0);
    var left = nextAt - (v.rd || 0);
    var th = Math.max(1, (s.int || 500) / 10);
    if (left <= th) {
      list.push({
        type: "s", id: s.id, vid: s.vid, lb: s.lb, left: left, u: u,
        lv: left < 0 ? 0 : 1,
        msg: s.lb + (left < 0 ? " overdue by " + (-left) : " due in " + left) + " " + u
      });
    }
  });

  return list.sort(function(a, b){ return a.lv - b.lv; });
}

function handleSearch(q) {
  var el = document.getElementById("searchRes");
  if (!el) return;
  q = (q || "").trim().toLowerCase();
  if (!q) { el.innerHTML = ""; return; }

  var hits = DB.V.filter(function(v){
    return (v.no||"").toLowerCase().indexOf(q) >= 0 || (v.nm||"").toLowerCase().indexOf(q) >= 0 || (v.f||"").toLowerCase().indexOf(q) >= 0;
  }).slice(0, 5);

  if (!hits.length) {
    el.innerHTML = '<div class="mu" style="padding:6px">No vehicle found.</div>';
    return;
  }

  var out = "";
  for (var i = 0; i < hits.length; i++) {
    var v = hits[i];
    out += '<div class="c" style="margin:4px 0;cursor:pointer" onclick="openVehicle(\'' + v.id + '\')">' +
      '<div class="r"><span class="plate ' + (v.c=="Truck"?"yl":"") + '"><i>IND</i><b>' + esc(v.no) + '</b></span><span class="mu sp">' + esc(v.nm) + ' · ' + esc(v.c) + ' · ' + esc(v.f||"") + '</span><span>›</span></div>' +
    '</div>';
  }
  el.innerHTML = out;
}

function saveNewVehicle() {

  var c = document.getElementById("new_vc").value;

  var rd = +document.getElementById("new_vrd").value || 0;

function resolveItem(type, id) {
  if (type === "d") {
    var paper = null;
    for (var i = 0; i < DB.D.length; i++) { if (DB.D[i].id == id) paper = DB.D[i]; }
    if (!paper) return;

    var nextYear = new Date();
    nextYear.setFullYear(nextYear.getFullYear() + 1);

    showModal('<b>Renew Paper: ' + esc(paper.k) + '</b><div class="mu" style="margin:6px 0">Select new validity expiry date:</div>' +
      '<input id="rnw_val" type="date" value="' + fmtDate(nextYear) + '">' +
      '<div class="r" style="margin-top:10px">' +
        '<button class="b" onclick="savePaperRenewal(\'' + id + '\')">Update Expiry</button>' +
        '<button class="s" onclick="closeModal()">Cancel</button>' +
      '</div>');
  } else if (type === "e") {
    for (var j = 0; j < DB.E.length; j++) {
      if (DB.E[j].id == id) {
        DB.E[j].due = addMonth(DB.E[j].due);
        break;
      }
    }
    persist();
    render();
  } else if (type === "s") {
    for (var k = 0; k < DB.Sv.length; k++) {
      if (DB.Sv[k].id == id) {
        var v = getVeh(DB.Sv[k].vid);
        DB.Sv[k].last = v.rd || 0;
        break;
      }
    }
    persist();
    render();
  }
}

function savePaperRenewal(id) {

  for (var i = 0; i < DB.D.length; i++) {
    if (DB.D[i].id == id) {
      DB.D[i].ex = val;
      break;
    }
  }
  persist();
  closeModal();
  render();
}

function buildReportData() {
  var dt = offsetDate(0);
  var out = { text: "", title: "", headers: [], rows: [] };

  if (state.repType === "e") {
    out.title = "EMI OUTLAYS REPORT";
    out.headers = ["#", "Vehicle", "Financer", "Debited Bank / Account", "Amount"];
    var list = DB.E.filter(function(e){ return diffDays(e.due) <= 30; }).sort(function(a,b){ return a.due.localeCompare(b.due); });

    if (state.repOnlyNext && list.length) {
      var firstDate = list[0].due;
      list = list.filter(function(e){ return e.due === firstDate; });
    }

    var total = list.reduce(function(acc, x){ return acc + (+x.amt || 0); }, 0);
    out.text = "*" + out.title + "*\nAs on: " + dt + "\nTotal EMI Amount: " + inr(total) + "\n\n";

    for (var i = 0; i < list.length; i++) {
      var e = list[i];
      var v = getVeh(e.vid);
      out.rows.push([i + 1, v.no, e.ln, e.bk || "-", inr(e.amt)]);
      out.text += (i + 1) + ". *" + v.no + "*\n   Financer: " + e.ln + "\n   Debited: " + (e.bk || "-") + "\n   Amount: " + inr(e.amt) + " (Due: " + e.due + ")\n\n";
    }
    out.text += "*Total Liabilities: " + inr(total) + "*";
  } else {
    out.title = "PAPERS EXPIRY REPORT";
    out.headers = ["#", "Vehicle", "Document", "Expiry Date"];
    var pList = DB.D.filter(function(d){ return d.ex && diffDays(d.ex) <= 30; }).sort(function(a,b){ return a.ex.localeCompare(b.ex); });

    out.text = "*" + out.title + "*\nAs on: " + dt + "\nTotal Papers: " + pList.length + "\n\n";
    for (var j = 0; j < pList.length; j++) {
      var dItem = pList[j];
      var vItem = getVeh(dItem.vid);
      out.rows.push([j + 1, vItem.no, dItem.k, dItem.ex]);
      out.text += (j + 1) + ". *" + vItem.no + "* - " + dItem.k + " - Expiry: " + dItem.ex + (diffDays(dItem.ex) < 0 ? " (EXPIRED)" : "") + "\n";
    }
  }

  return out;
}

function generateReportImage() {
  var data = buildReportData();
  var W = 720;
  var rowH = 32;
  var H = 140 + (data.rows.length + 1) * rowH;

  var canvas = document.createElement("canvas");
  canvas.width = W;
  canvas.height = H;
  var ctx = canvas.getContext("2d");

  ctx.fillStyle = "#ffffff";
  ctx.fillRect(0, 0, W, H);
  ctx.fillStyle = "#0b6e4f";
  ctx.fillRect(0, 0, W, 70);

  ctx.fillStyle = "#ffffff";
  ctx.font = "bold 20px sans-serif";
  ctx.fillText(data.title, 20, 32);
  ctx.font = "12px sans-serif";
  ctx.fillText("Generated on: " + offsetDate(0), 20, 54);

  var y = 95;
  var colWidths = (data.headers.length === 5) ? [35, 140, 170, 230, 105] : [40, 200, 260, 180];

  function drawRow(cells, isHead) {
    ctx.fillStyle = isHead ? "#24303e" : "#ffffff";
    ctx.fillRect(20, y, W - 40, rowH);
    ctx.strokeStyle = "#e2e8f0";
    ctx.strokeRect(20, y, W - 40, rowH);

    ctx.fillStyle = isHead ? "#ffffff" : "#111827";
    ctx.font = (isHead ? "bold 12px " : "12px ") + "sans-serif";

    var cx = 25;
    for (var i = 0; i < cells.length; i++) {
      ctx.fillText(String(cells[i]), cx, y + 21);
      cx += colWidths[i];
    }
    y += rowH;
  }

  drawRow(data.headers, true);
  for (var r = 0; r < data.rows.length; r++) {
    drawRow(data.rows[r], false);
  }

  var imgUrl = canvas.toDataURL("image/png");
  showModal('<b>Report Image Generated</b><div class="mu" style="margin:6px 0">Download PNG file directly to your phone storage:</div>' +
    '<img src="' + imgUrl + '" style="width:100%;border:1px solid var(--bd);border-radius:8px">' +
    '<div class="r" style="margin-top:10px">' +
      '<a class="b" download="Fleet_Report_' + offsetDate(0) + '.png" href="' + imgUrl + '">Download Image (PNG)</a>' +
      '<button class="s" onclick="closeModal()">Close</button>' +
    '</div>');
}

function restoreBackup(file) {
  if (!file) return;
  var reader = new FileReader();
  reader.onload = function() {
    try {
      var parsed = JSON.parse(reader.result);
      if (Array.isArray(parsed.V)) DB.V = parsed.V;
      if (Array.isArray(parsed.D)) DB.D = parsed.D;
      if (Array.isArray(parsed.E)) DB.E = parsed.E;
      if (Array.isArray(parsed.Sv)) DB.Sv = parsed.Sv;
      if (Array.isArray(parsed.FIRMS)) DB.FIRMS = parsed.FIRMS;
      persist();
      render();
      alert("Backup successfully restored!");
    } catch(err) {
      alert("Invalid backup file.");
    }
  };
  reader.readAsText(file);
}

function esc(s) {
  return String(s == null ? "" : s).replace(/[&<>"]/g, function(c){
    return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c];
  });
}
